# 🔄 Switching from OpenAI to Local/Alternative AI Models

I understand you want to use an alternative to OpenAI's API. Let me show you how to modify your agents to use different AI services or local models. This is a great approach that can save costs and give you more control.

## Option 1: Using Free Alternative APIs

Let's modify the agents to use free alternatives like Anthropic's Claude, Google's Gemini, or local models:

### Step 2: Create a Universal AI Helper

Create a new file `ai_helper.py` with this versatile AI integration:

```python
# ai_helper.py
import os
import requests
import json
from typing import Dict, Any

class AIHelper:
    def __init__(self, provider="openai"):
        self.provider = provider
        self.setup_providers()
    
    def setup_providers(self):
        """Setup configuration for different AI providers"""
        self.providers = {
            "openai": {
                "url": "https://api.openai.com/v1/chat/completions",
                "headers": {
                    "Authorization": f"Bearer {os.getenv('OPENAI_API_KEY', '')}",
                    "Content-Type": "application/json"
                },
                "payload": {
                    "model": "gpt-3.5-turbo",
                    "temperature": 0.3
                }
            },
            "anthropic": {
                "url": "https://api.anthropic.com/v1/messages",
                "headers": {
                    "x-api-key": os.getenv('ANTHROPIC_API_KEY', ''),
                    "anthropic-version": "2023-06-01",
                    "Content-Type": "application/json"
                },
                "payload": {
                    "model": "claude-3-sonnet-20240229",
                    "max_tokens": 1000
                }
            },
            "google": {
                "url": "https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent",
                "headers": {
                    "Content-Type": "application/json"
                },
                "payload": {
                    "temperature": 0.3
                }
            },
            "huggingface": {
                "url": "https://api-inference.huggingface.co/models/mistralai/Mistral-7B-Instruct-v0.2",
                "headers": {
                    "Authorization": f"Bearer {os.getenv('HF_API_KEY', '')}"
                }
            },
            "local": {
                "url": "http://localhost:11434/api/generate",  # For Ollama
                "headers": {
                    "Content-Type": "application/json"
                },
                "payload": {
                    "model": "llama2",
                    "stream": False
                }
            }
        }
    
    def generate_response(self, prompt: str, system_message: str = None) -> str:
        """Generate a response using the configured AI provider"""
        try:
            if self.provider == "openai":
                return self._call_openai(prompt, system_message)
            elif self.provider == "anthropic":
                return self._call_anthropic(prompt, system_message)
            elif self.provider == "google":
                return self._call_google(prompt, system_message)
            elif self.provider == "huggingface":
                return self._call_huggingface(prompt)
            elif self.provider == "local":
                return self._call_local(prompt)
            else:
                raise ValueError(f"Unsupported provider: {self.provider}")
        except Exception as e:
            print(f"Error calling {self.provider}: {str(e)}")
            return f"AI service error: {str(e)}"
    
    def _call_openai(self, prompt: str, system_message: str = None) -> str:
        """Call OpenAI API"""
        config = self.providers["openai"]
        messages = []
        
        if system_message:
            messages.append({"role": "system", "content": system_message})
        messages.append({"role": "user", "content": prompt})
        
        payload = config["payload"].copy()
        payload["messages"] = messages
        
        response = requests.post(
            config["url"],
            headers=config["headers"],
            json=payload
        )
        response.raise_for_status()
        return response.json()["choices"][0]["message"]["content"]
    
    def _call_anthropic(self, prompt: str, system_message: str = None) -> str:
        """Call Anthropic API"""
        config = self.providers["anthropic"]
        
        messages = [{"role": "user", "content": prompt}]
        
        payload = config["payload"].copy()
        payload["messages"] = messages
        
        if system_message:
            payload["system"] = system_message
        
        response = requests.post(
            config["url"],
            headers=config["headers"],
            json=payload
        )
        response.raise_for_status()
        return response.json()["content"][0]["text"]
    
    def _call_google(self, prompt: str, system_message: str = None) -> str:
        """Call Google Gemini API"""
        config = self.providers["google"]
        
        # Add API key to URL for Google
        url = f"{config['url']}?key={os.getenv('GOOGLE_API_KEY', '')}"
        
        contents = []
        if system_message:
            contents.append({"role": "user", "parts": [{"text": system_message}]})
        contents.append({"role": "user", "parts": [{"text": prompt}]})
        
        payload = {"contents": contents}
        
        response = requests.post(
            url,
            headers=config["headers"],
            json=payload
        )
        response.raise_for_status()
        return response.json()["candidates"][0]["content"]["parts"][0]["text"]
    
    def _call_huggingface(self, prompt: str) -> str:
        """Call Hugging Face API"""
        config = self.providers["huggingface"]
        
        payload = {"inputs": prompt}
        
        response = requests.post(
            config["url"],
            headers=config["headers"],
            json=payload
        )
        response.raise_for_status()
        return response.json()[0]["generated_text"]
    
    def _call_local(self, prompt: str) -> str:
        """Call local Ollama API"""
        config = self.providers["local"]
        
        payload = config["payload"].copy()
        payload["prompt"] = prompt
        
        response = requests.post(
            config["url"],
            headers=config["headers"],
            json=payload
        )
        response.raise_for_status()
        return response.json()["response"]

# Test the AI helper
def test_ai_helper():
    """Test the AI helper with different providers"""
    helper = AIHelper(provider="openai")  # Change to your preferred provider
    
    test_prompt = "What is the capital of France?"
    response = helper.generate_response(test_prompt)
    print(f"AI Response: {response}")

if __name__ == "__main__":
    test_ai_helper()
```

### Step 6: Option - Use Local Models with Ollama

If you want to use completely local models without API costs:

1. Install Ollama: https://ollama.ai/

3. Set your `.env` to use local provider:

## Benefits of This Approach

1. **Cost Savings**: Use free tiers or local models to avoid API costs
2. **Flexibility**: Easily switch between different AI providers
3. **Redundancy**: If one service is down, you can switch to another
4. **Privacy**: Local models keep your data on your own machine
5. **Customization**: You can fine-tune local models for your specific needs
